1"""Guardrails — refuse-when-unsupported, groundedness, and PII redaction. 

2 

3These are what separate a RAG demo from a RAG system you'd put in front of a 

4client: the model is not allowed to answer from nothing, its answer is checked 

5against the retrieved context, and PII is stripped from the output. Deterministic 

6and stdlib so each guard is unit-testable; the groundedness proxy is swappable 

7for an NLI/LLM judge behind the same interface. 

8""" 

9from __future__ import annotations 

10 

11import re 

12 

13_TOKEN = re.compile(r"[a-z0-9]{2,}") 

14_STOP = { 

15 "the", "and", "for", "are", "was", "you", "your", "our", "with", "that", "this", 

16 "from", "have", "has", "can", "will", "any", "all", "out", "use", "via", "per", 

17 "is", "to", "of", "in", "on", "at", "it", "or", "be", "we", "a", "an", "as", "by", 

18 "takes", "take", "made", "business", "days", 

19} 

20 

21_EMAIL = re.compile(r"\b[\w.+-]+@[\w-]+\.[\w.-]+\b") 

22_PHONE = re.compile(r"\b\d{3}[-.\s]\d{3}[-.\s]\d{4}\b") 

23_SSN = re.compile(r"\b\d{3}-\d{2}-\d{4}\b") 

24_CARD = re.compile(r"\b(?:\d[ -]?){13,16}\b") 

25 

26 

27def _content_tokens(text: str) -> set[str]: 

28 return {t for t in _TOKEN.findall((text or "").lower()) if t not in _STOP} 

29 

30 

31def should_refuse(hits: list[dict], min_score: float = 0.05) -> bool: 

32 """Refuse to answer when retrieval found no real support (top score too low).""" 

33 if not hits: 

34 return True 

35 return max(h.get("score", 0.0) for h in hits) < min_score 

36 

37 

38def groundedness(answer: str, contexts: list[str], threshold: float = 0.5) -> dict: 

39 """What fraction of the answer's content words are supported by the context. 

40 `grounded` is support >= threshold. (Lexical proxy for an NLI/LLM judge.)""" 

41 ans = _content_tokens(answer) 

42 if not ans: 

43 return {"grounded": False, "support": 0.0} 

44 ctx = set() 

45 for c in contexts: 

46 ctx |= _content_tokens(c) 

47 support = len(ans & ctx) / len(ans) 

48 return {"grounded": support >= threshold, "support": round(support, 4)} 

49 

50 

51def redact_pii(text: str) -> str: 

52 """Mask emails, phone numbers, SSNs, and card-like numbers in an output.""" 

53 text = _EMAIL.sub("[redacted-email]", text) 

54 text = _SSN.sub("[redacted-ssn]", text) 

55 text = _PHONE.sub("[redacted-phone]", text) 

56 text = _CARD.sub("[redacted-card]", text) 

57 return text
